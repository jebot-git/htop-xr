cmake_minimum_required(VERSION 3.24)
project(htop-fltk LANGUAGES C CXX)
include(GNUInstallDirs)
find_package(FLTK 1.4.5 EXACT CONFIG REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(FONTS REQUIRED IMPORTED_TARGET fontconfig zlib)
find_library(NCURSES ncursesw REQUIRED)
find_library(TINFO tinfo REQUIRED)
find_library(CAP cap REQUIRED)
find_library(UNWIND unwind REQUIRED)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(collector STATIC Collector.c)
target_include_directories(collector PRIVATE "${ROOT}/build/tui" "${ROOT}" "${ROOT}/linux")
target_compile_features(collector PRIVATE c_std_99)
target_compile_definitions(collector PRIVATE SYSCONFDIR="/usr/local/etc")
add_executable(htop-fltk Main.cpp)
target_compile_features(htop-fltk PRIVATE cxx_std_17)
target_compile_options(htop-fltk PRIVATE -Wall -Wextra)
target_link_libraries(htop-fltk PRIVATE collector "${ROOT}/build/tui/libhtop-collector.a" ${CAP} ${UNWIND} ${NCURSES} ${TINFO} m fltk::fltk-shared fltk::gl-shared fltk::images-shared PkgConfig::FONTS)
set_target_properties(htop-fltk PROPERTIES INSTALL_RPATH "$<TARGET_FILE_DIR:fltk::fltk-shared>;$<TARGET_FILE_DIR:fltk::gl-shared>")
include(desktop-menu.cmake)
de_xr_desktop_menu(htop-fltk)
install(TARGETS htop-fltk RUNTIME DESTINATION libexec/htop)
include(CTest)
if(BUILD_TESTING)
  add_executable(collector-test CollectorTest.cpp)
  target_compile_features(collector-test PRIVATE cxx_std_17)
  target_compile_options(collector-test PRIVATE -UNDEBUG)
  target_link_libraries(collector-test PRIVATE collector "${ROOT}/build/tui/libhtop-collector.a" ${CAP} ${UNWIND} ${NCURSES} ${TINFO} m PkgConfig::DESKTOPMENU)
  add_test(NAME upstream-collector-and-process-actions COMMAND collector-test)
endif()
